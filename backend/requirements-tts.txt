# Optional: Chatterbox TTS (Resemble AI) — voice cloning across 23 languages.
#
# WHY THIS IS A SEPARATE FILE
# Chatterbox pins a newer ML stack than the MuseTalk lip-sync pipeline in
# requirements.txt can run against:
#
#              requirements.txt   chatterbox-tts 0.1.7
#   torch            2.2.0                2.6.0
#   torchaudio       2.2.0                2.6.0
#   transformers     4.37.2               5.2.0
#   librosa          0.10.1               0.11.0
#
# Those are mutually exclusive, so the two cannot share one environment.
# Installing this file over requirements.txt WILL break MuseTalk.
#
# HOW TO USE IT
# Give Chatterbox its own virtualenv and run it as a separate TTS service, or
# accept that this environment does lip-sync OR cloned-voice TTS, not both:
#
#   python -m venv venv-tts
#   ./venv-tts/bin/pip install -r requirements-tts.txt
#   ./venv-tts/bin/python tts_server.py          # then CHATTERBOX_URL=http://127.0.0.1:8001
#
# WITHOUT IT
# Nothing breaks. Without CHATTERBOX_URL, app/services/tts.py tries an
# in-process import, and falls through to Edge TTS (free Microsoft neural
# voices, CPU-only) and then gTTS. Voice cloning is the only feature you lose.

chatterbox-tts==0.1.7

# Pinned to match what chatterbox-tts 0.1.7 itself requires, so this file
# resolves on its own rather than relying on pip to backtrack.
torch==2.6.0
torchaudio==2.6.0
transformers==5.2.0
librosa==0.11.0

# HTTP wrapper (tts_server.py) so the backend can call this venv remotely.
fastapi>=0.110
